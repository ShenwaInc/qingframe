## 客户端示例
### uni-app 调用 SwaSocket 信令服务
```vue
import SwaSocket from "@/utils/swa-socket.js"

  const userSign = '7ac93049cfaa1f7f3b8eb92477cc577a';	//用户标识符，通过服务端生成
  const server = 'wss://socket.whotalk.com.cn/wss';			//socket服务地址，通过服务端获得

  const handleMessage = function(data){
  // 接收到 SOCKET 消息事件
  console.log(data);
  }

  SwaSocket.socketHost = 'https://whotalk.baidu.com/'; //对接域名，用于验证授权
  SwaSocket.onDisconnect = (res)=>{
  // 监听 SOCKET 断开连接事件
  console.log(res);
  //{code: 10000, reason: ""}
  }
  SwaSocket.onHeartbeat = ()=>{
  // 监听 SOCKET 心跳事件
  }
  SwaSocket.onConnect = ()=>{
  // 监听 SOCKET 连接成功事件
  }
  SwaSocket.init(userSign, server, handleMessage, (err)=>{
  // SOCKET 连接中断
  });
```

```html
        window.Swaws.init(userSign, server, (res)=>{

            console.log(res);
            if(res.type==="openDoor"){
                //执行开门请求，门锁ID是 res.data.id
                //无论开门成功或者失败都触发设备数据上报
            }
        });
    }
</script>
```
