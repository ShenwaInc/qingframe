调用示例



```html
<template>
  <view>
    <view class="btn btn-primary" @click="showPicker=true">
      <text>选择地区</text>
    </view>
    <citypicker 
      :ShowPicker="showPicker"
      :province="selectedProvince"
      :city="selectedCity"
      :dist="selectedDist"
      :fields="dist"
      :title="请选择地区"
      :textbtn="确认选择"
      :textclear="清除"
      @onCancel="onPickerCancel"
      @onClear="onPickerClear"
      @onResult="onPickerResult">
    </citypicker>
  </view>
</template>

<script>
  import citypicker from "@/components/citypicker.vue"
  
  export default {
    components: { citypicker },
    data() {
      return {
        showPicker: false,
        selectedProvince: "广东省",
        selectedCity: "深圳市",
        selectedDist: "南山区"
      }
    },
    methods: {
      onPickerCancel() {
        this.showPicker = false;
        console.log('取消选择');
      },
      onPickerClear() {
        this.selectedProvince = "";
        this.selectedCity = "";
        this.selectedDist = "";
        this.showPicker = false;
        console.log('清除选择');
      },
      onPickerResult(data) {
        this.selectedProvince = data.province;
        this.selectedCity = data.city;
        this.selectedDist = data.dist;
```
